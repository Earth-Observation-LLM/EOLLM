\begin{frame}{The question}
\large
Each urban VQA question shows \textbf{4 street-view cameras}
(\texttt{fwd}, \texttt{bwd}, \texttt{cross\_left}, \texttt{cross\_right}).
\vspace{6pt}
\begin{block}{Hypothesis}
The full 360\textdegree{} view is \textbf{redundant}: the \textbf{one image the model
attends to most} keeps the answer correct --- and that image is the
\textbf{forward camera}.
\end{block}
\vspace{6pt}
\footnotesize
\textbf{Method.} Solve each question on all 4 images $\to$ keep only the single
chosen image $\to$ re-ask greedily. \textbf{Survival \%} $=$ fraction of solved
questions still correct on the kept image.
\par\vspace{3pt}
\textbf{Arms:} \textcolor{ours}{\textbf{top}} (most-attended) $\,\vert\,$
\textcolor{fwdc}{\textbf{fwd}} (always forward) $\,\vert\,$
random $\,\vert\,$ \textcolor{botc}{\textbf{bottom}} (least-attended).
\end{frame}

\begin{itemize}\footnotesize
\item Attention's \#1 image \textbf{is the forward camera in 94.8\%} of questions.
\item Pooled survival: \textcolor{ours}{top} $89.5\%$ $\approx$
 \textcolor{fwdc}{fwd} $89.1\%$ --- picking by attention and ``always forward''
 land in the same place.
\item Practical rule: \textbf{keep the forward image, drop the other three}.
\end{itemize}

\begin{frame}{Take-home}
\Large
\begin{itemize}
\item[\textcolor{good}{\ding{51}}] The 360\textdegree{} view is \textbf{redundant} ---
 \textbf{one} street-view image suffices ($89.5\%$ survival).
\item[\textcolor{good}{\ding{51}}] That image is the \textbf{forward camera}
 (\texttt{along\_fwd}), chosen by attention $94.8\%$ of the time.
\item[\textcolor{good}{\ding{51}}] The right image \textbf{beats a random one}
 by $+7$ pts, in \textbf{every} topic.
\end{itemize}
